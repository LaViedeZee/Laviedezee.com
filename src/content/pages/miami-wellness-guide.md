(function() {
  var ANCHORS = {
    'Float Tanks & Sensory Deprivation': '#float-section',
    'Indigenous & Earth Healing': '#earth-healing-section',
    'Light Therapy': '#light-therapy-section',
    'Meditation': '#meditation-section',
    'Reiki & Energy Work': '#reiki-section',
    'Salt Rooms (Halotherapy)': '#salt-room-section',
    'Somatic Bodywork & Breathwork': '#somatic-section',
    'Sound Healing & Sound Baths': '#sound-healing-section',
    'Thermal & Contrast Therapy': '#thermal-contrast-section',
    'Water Therapy (Watsu & Hydrotherapy)': '#water-therapy-section'
  };

  var table = document.querySelector('.wellness-table');
  if (!table) return;

  var headers = Array.from(table.querySelectorAll('thead th')).map(function(th) {
    return th.textContent.trim();
  });
  var neighborhoods = headers.slice(1);

  var rows = Array.from(table.querySelectorAll('tbody tr'));
  var modalities = rows.map(function(row) {
    return row.querySelector('td').textContent.trim();
  });

  var data = [];
  rows.forEach(function(row, ri) {
    var tds = Array.from(row.querySelectorAll('td'));
    neighborhoods.forEach(function(hood, hi) {
      var cellEl = tds[hi + 1];
      if (!cellEl) return;
      var ta = cellEl.querySelector('textarea');
      var text = ta ? ta.value.trim() : cellEl.textContent.trim();
      if (!text) return;
      var spaces = text.split('\n').map(function(s) { return s.trim(); }).filter(Boolean);
      spaces.forEach(function(space) {
        data.push({ name: space, modality: modalities[ri], neighborhood: hood });
      });
    });
  });

  var modSelect = document.getElementById('be-modality');
  var hoodSelect = document.getElementById('be-neighborhood');
  var resultsEl  = document.getElementById('be-results');

  if (!modSelect || !hoodSelect) return;

  modalities.slice().sort().forEach(function(m) {
    var o = document.createElement('option');
    o.value = m; o.textContent = m;
    modSelect.appendChild(o);
  });

  neighborhoods.slice().sort().forEach(function(h) {
    var o = document.createElement('option');
    o.value = h; o.textContent = h;
    hoodSelect.appendChild(o);
  });

  function render() {
    var mod = modSelect.value;
    var hood = hoodSelect.value;

    if (!mod && !hood) {
      resultsEl.innerHTML = '<p class="be-prompt">Select a modality or neighborhood to explore wellness spaces.</p>';
      return;
    }

    var filtered = data.filter(function(item) {
      return (!mod || item.modality === mod) && (!hood || item.neighborhood === hood);
    });

    var seen = {};
    var unique = filtered.filter(function(item) {
      var key = item.name + '|' + item.neighborhood + '|' + item.modality;
      if (seen[key]) return false;
      seen[key] = true;
      return true;
    });

    if (!unique.length) {
      resultsEl.innerHTML = '<p class="be-no-results">No wellness spaces found for that combination.</p>';
      return;
    }

    unique.sort(function(a, b) { return a.name.localeCompare(b.name); });

    resultsEl.innerHTML = unique.map(function(item) {
      var anchor = ANCHORS[item.modality] || '#wellness-spaces';
      return '<a href="' + anchor + '" class="be-card">' +
        '<p class="be-card-name">' + item.name + '</p>' +
        '<p class="be-card-meta">' + item.neighborhood + '</p>' +
        '<div class="be-card-footer">' +
          '<div class="be-card-tags"><span class="be-tag">' + item.modality + '</span></div>' +
          '<span class="be-card-arrow">View spaces ↓</span>' +
        '</div>' +
        '</a>';
    }).join('');
  }

  modSelect.addEventListener('change', render);
  hoodSelect.addEventListener('change', render);
